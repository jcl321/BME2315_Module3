import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import curve_fit

def Fact(L1, L2, Kd1, Kd2):
    """Fractional receptor activation with two agonists L1 and L2"""
    return (Kd2*L1+Kd1*L2) / (Kd1*Kd2 + Kd2*L1 + Kd1*L2)

# Parameters [nM]
Kd1 = 0.17 # L1, GLP1
L1 = 0.015 # [nM] GLP1 (fasting)
L2 = np.logspace(-3, 2, 500) # [nM] cavalitide

# Load the data
data = np.loadtxt(r"Module 3\BME2315_Module3\cavalitide_data.csv", delimiter=",", skiprows=1)
L2_loaded, Fact_loaded = data[:, 0], data[:, 1]

# Approach 1: Brute force optimization of Kd2: try many candidate values in a for loop
Kd2_candidates = np.logspace(-3, 2, 100) # [nM] # checks 100 values between 10^-3 and 10^2 for minimized error
SSE_brute = np.inf
for Kd2_try in Kd2_candidates:
    # Sum of squared errors between the data and the model for this candidate
    SSE = np.sum((Fact_loaded - Fact(L1, L2_loaded, Kd1, Kd2_try))**2)
    if SSE < SSE_brute:
        # stores square of errors and kd2 value if it is lower than the previous minimum
        SSE_brute = SSE 
        Kd2_brute = Kd2_try
print(f"Brute-force Kd2 = {Kd2_brute:.3f} nM (SSE = {SSE_brute:.4f})")

# Approach 2: Python curvefit of Kd2: related to steepest descent method
popt, pcov = curve_fit(lambda L2, Kd2: Fact(L1, L2, Kd1, Kd2),
                       L2_loaded, Fact_loaded, p0=[0.1], bounds=(0, np.inf))
Kd2_fit = popt[0]
SSE_fit = np.sum((Fact_loaded - Fact(L1, L2_loaded, Kd1, Kd2_fit))**2)
print(f"Fitted Kd2 = {Kd2_fit:.3f} nM (SSE = {SSE_fit:.4f})")

# product-specific parameters
products = {

                    ka=0.029,          # 1/h, absorption rate
                    D=2.0,             # mg per dose

    # NEW (Task 2.2): Cavalitide pill, same as Wegovy but faster absorption
    'Cavalitide': dict(Favailable=0.01,
                    ka=0.006,
                    D=25,
                    interval=24,
                    n_doses=100,
                    depot='gut (oral)'),
    # NEW (Task 2.3): Cavalitide with lower dose to reach ~0.5 nM
    'Cavalitide 21 mg': dict(Favailable=0.01,
                    ka=0.006,
                    D=21,
                    interval=24,
                    n_doses=100,
                    depot='gut (oral)'),
}

# shared model parameters
ke = 0.0039     # 1/h, elimination rate
V = 10.8        # L
MW = 4113.6     # g/mol
Ffree = 0.01    # rest of semaglutide bound to albumin

# simulation parameters
dt = 0.1        # h, spacing of output time points

def ODEfunc(t, y, ka):
    Asc, Cc = y
    dAsc = -ka*Asc            # Asc: depot amount (mg)
    dCc = ka*Asc/V - ke*Cc    # Cc: blood concentration (mg/L)
    return [dAsc, dCc]

def simulate(Favailable, ka, D, n_doses, interval, dt):
    """Repeated dosing every `interval` h; returns time (h), depot amount (mg),
    and blood concentration (nmol/L)."""
    t_out, Asc_out, Cc_out = [], [], []
    y = np.array([0.0, 0.0])
    n_steps = int(round(interval / dt))
    for i in range(n_doses):
        y[0] += Favailable*D                 # absorbed fraction of dose enters the depot
        t0 = i * interval
        t_seg = t0 + dt * np.arange(n_steps + 1)

        # CHANGED: solve_ivp replaces the explicit Euler loop
        sol = solve_ivp(ODEfunc, (t_seg[0], t_seg[-1]), y, t_eval=t_seg,
                        args=(ka,), rtol=1e-6)
        y_seg = sol.y

        t_out.append(t_seg)
        Asc_out.append(y_seg[0])
        Cc_out.append(y_seg[1])
        y = y_seg[:, -1].copy()         # carry state into next interval
    t = np.concatenate(t_out)
    Asc_out = np.concatenate(Asc_out)
    Cc_out = np.concatenate(Cc_out)
    Cc_nM = Cc_out * 1e-3 / MW * 1e9
    return t, Asc_out, Cc_nM

# --- run and plot (CHANGED: plot several drugs together to compare) ---
def compare(drugs, title):
    for drug in drugs:
        p = products[drug]
        t, Asc, Cc = simulate(p['Favailable'], p['ka'], p['D'],

    plt.ylabel("Free semaglutide/Cavalitide in blood (nM)")

The pharmacokinetic model is then simulated for each drug with their given parameters. The solve_ivp python method is used to model a solution of the ODEs which describe the concentration of the drug in the blood and in the deposit (gut or subcutaneous) over time.

# ==========================================
# PHARMACOKINETIC MODEL
# ==========================================

def ODEfunc(t, y, ka):
    Asc, Cc = y
    dAsc = -ka*Asc            # Asc: depot amount (mg)
    dCc = ka*Asc/V - ke*Cc    # Cc: blood concentration (mg/L)
    return [dAsc, dCc]


def simulate(Favailable, ka, D, n_doses, interval, dt):
    """Repeated dosing every `interval` h; returns time (h), depot amount (mg),
    and blood concentration (nmol/L)."""
    t_out, Asc_out, Cc_out = [], [], []
    y = np.array([0.0, 0.0])
    n_steps = int(round(interval / dt))
    for i in range(n_doses):
        y[0] += Favailable*D                 # absorbed fraction of dose enters the depot
        t0 = i * interval
        t_seg = t0 + dt * np.arange(n_steps + 1)

        # solve_ivp solves differential equation to find solution (blood concentration over time)
        sol = solve_ivp(ODEfunc, (t_seg[0], t_seg[-1]), y, t_eval=t_seg,
                        args=(ka,), rtol=1e-6)
        y_seg = sol.y

        t_out.append(t_seg)
        Asc_out.append(y_seg[0])
        Cc_out.append(y_seg[1])
        y = y_seg[:, -1].copy()         # carry state into next interval
    t = np.concatenate(t_out)
    Asc_out = np.concatenate(Asc_out)
    Cc_out = np.concatenate(Cc_out)
    Cc_nM = Cc_out * 1e-3 / MW * 1e9
    return t, Asc_out, Cc_nM

Limitations and Future Work:¶

The results are only computational models, and are therefore limited due to potential for variation to occur in real life. Values provided are only estimates using computer methods and do not perfectly model behavior of cavalitide or semaglutide. Future work would require clinical trials on a variety of patients to observe if data trends are visible across multiple samples. Additionally, the model only observes data of cavalitide interaction with GLP1 receptors. It does not explore potential for cavalitide to act as an agonist for other receptors and result in side effects. Future work should analyze pathways to assure that administering of cavalitide does not result in any dangerous adverse effects. The data does not directly relate the drug's behavior to weight loss, therefore future work should study if a directly causational relationship can be found between cavalitide usage and weight loss. Additionally, the viability of the drug in the market must be analyzed before it can be implemented into the public. A cost analysis and marketing study should assure that a pharmaceutical company would be able to offer the drug to the public and it can successfully compete with existing weight loss drugs. In order to ethically develop the drug, potential side effects should be studied before clinical trials on human patients occur. Patients should be informed that long-term risks of the drug are not yet fully understood to assure informed consent. Although early results seem promising, additional work is necessary to determine success of drug implementation.
